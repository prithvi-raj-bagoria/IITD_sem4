// Test case for control flow statements

int x := 10;
int y := 20;

// If-else statement
if (x > y) {
    print("x is greater than y");
} else {
    print("y is greater than or equal to x");
}

// Nested if statement
if (x == 10) {
    if (y == 20) {
        print("x is 10 and y is 20");
    }
}

// While loop
int counter := 0;
while (counter < 5) {
    print(counter);
    counter := counter + 1;
}

// Do-while loop
int j := 5;
do {
    print(j);
    j := j - 1;
} while (j > 0);

// For loop
for (int i := 0; i < 3; i := i + 1) {
    print(i);
}

// Nested loops
for (int i := 1; i <= 3; i := i + 1) {
    for (int j := 1; j <= i; j := j + 1) {
        print(i * j);
    }
}
