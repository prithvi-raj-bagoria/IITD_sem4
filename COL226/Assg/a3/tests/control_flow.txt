// Control flow test cases

// If-else statements with complex conditions
int x := 10;
int y := 5;

if (x > y && x * y > 25 || x - y < 4) {
    print(1);  // Should not execute
} else {
    print(0);  // Should execute
}

// Nested if statements
if (x > 0) {
    if (y > 0) {
        print(1);  // Should execute
    } else {
        print(2);  // Should not execute
    }
} else {
    print(3);  // Should not execute
}

// For loops with complex initialization and update
int sum := 0;
for (int i := 1; i <= 10; i := i + 1) {
    sum := sum + i;
}
print(sum);  // 55

// Nested loops
int matrix_sum := 0;
for (int i := 0; i < 3; i := i + 1) {
    for (int j := 0; j < 3; j := j + 1) {
        matrix_sum := matrix_sum + (i * 3 + j + 1);
    }
}
print(matrix_sum);  // Sum of 1-9 = 45

// While loops with complex conditions
int n := 20;
int fib1 := 0;
int fib2 := 1;
while (fib2 < n) {
    int temp := fib2;
    fib2 := fib1 + fib2;
    fib1 := temp;
}
print(fib2);  // First Fibonacci number >= 20 (21)

// Break-like behavior using conditional
int prime := 17;
int divisor := 2;
bool is_prime := true;
while (divisor * divisor <= prime && is_prime) {
    if (prime % divisor == 0) {
        is_prime := false;
    }
    divisor := divisor + 1;
}
print(is_prime);  // true for 17
