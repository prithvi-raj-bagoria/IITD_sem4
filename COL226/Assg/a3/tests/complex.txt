// Complex program combining multiple language features

// Function to calculate factorial using recursion simulation
int n := 6;
int factorial := 1;
int i := 1;
while (i <= n) {
    factorial := factorial * i;
    i := i + 1;
}
print(factorial);  // 720

// Matrix operations for a simple linear transformation
vector v := [3, 4];
matrix rotation := [[0, -1], [1, 0]];  // 90-degree rotation matrix
matrix scaling := [[2, 0], [0, 2]];    // Scaling matrix

// Apply transformations: first rotate, then scale
matrix combined := scaling * rotation;
// No direct matrix-vector multiplication, so do it manually
float v_new_x := combined[0][0] * v[0] + combined[0][1] * v[1];
float v_new_y := combined[1][0] * v[0] + combined[1][1] * v[1];
vector v_transformed := [v_new_x, v_new_y];  // Should be [-8, 6]
print(v_transformed);

// Statistical calculations on a dataset
vector data := [12.5, 9.8, 15.2, 13.4, 11.7];
int data_size := dim(data);

// Calculate mean
float sum := 0.0;
for (int j := 0; j < data_size; j := j + 1) {
    sum := sum + data[j];
}
float mean := sum / data_size;
print(mean);

// Calculate variance
float variance_sum := 0.0;
for (int j := 0; j < data_size; j := j + 1) {
    float diff := data[j] - mean;
    variance_sum := variance_sum + diff * diff;
}
float variance := variance_sum / data_size;
print(variance);

// Calculate standard deviation
float std_dev := sqrt(variance);
print(std_dev);

// Find min and max
float min_val := data[0];
float max_val := data[0];
for (int j := 1; j < data_size; j := j + 1) {
    if (data[j] < min_val) {
        min_val := data[j];
    }
    if (data[j] > max_val) {
        max_val := data[j];
    }
}
print(min_val);
print(max_val);
