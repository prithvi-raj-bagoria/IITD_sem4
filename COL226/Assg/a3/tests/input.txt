vector input_vec_b;
input_vec_b := input();

// Solve using inverse: x = A^-1 * b
float det_A;
det_A := det(input_mat_A);
if (det_A != 0) {
    vector solution_x;
    solution_x := inverse(input_mat_A) * input_vec_b;

    print(solution_x);
    
    // Verify: A * x = b

    verification_result := input_mat_A * solution_x;
