// Advanced test case: Gaussian elimination algorithm

// Initialize a matrix
matrix 3,3 A;
A := 3,3 [[2, 1, 1], [4, 3, 3], [8, 7, 9]];

// Print the original matrix
print(A);

// Variables for Gaussian elimination
float pivot;
float factor;
int i;
int j;
int k;

// Forward elimination phase
for (i := 0; i < 3; i := i + 1) {
    // Get the pivot element
    pivot := A[i][i];
    
    // If pivot is zero, matrix is singular
    if (pivot == 0) {
        print("Matrix is singular, cannot proceed");
    } else {
        // Eliminate entries below pivot
        for (j := i + 1; j < 3; j := j + 1) {
            factor := A[j][i] / pivot;
            
            // Update all elements in the current row
            for (k := i; k < 3; k := k + 1) {
                A[j][k] := A[j][k] - factor * A[i][k];
            }
        }
    }
}

// Print the upper triangular form
print("Upper triangular form:");
print(A);
