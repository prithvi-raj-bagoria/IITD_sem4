// ===================== PRIMITIVE DATA TYPES & OPERATIONS =====================
print(1000);  // Section marker for primitive types

// Integer operations
int a;
a := 5;
print(a);

int b;
b := 7;
print(a + b);  // Addition: 12
print(a - b);  // Subtraction: -2
print(a * b);  // Multiplication: 35
print(b / a);  // Division: 1
print(b % a);  // Modulo: 2
print(a ** b); // Power: 78125

// Float operations
float x;
x := 3.14;
float y;
y := 2.71;
print(x + y);  // 5.85
print(x * y);  // 8.5094
print(x / y);  // 1.15866

// Mixed numeric operations
print(a + x);  // Int + Float: 8.14
print(b * y);  // Int * Float: 18.97
print(x / b);  // Float / Int: 0.44857

// Boolean operations
bool p;
p := true;
bool q;
q := false;
print(p && q); // AND: false
print(p || q); // OR: true
print(!p);     // NOT: false
print(p ^ q);  // XOR: true

// Comparison operations
print(a < b);   // Less than: true
print(a <= b);  // Less than or equal: true
print(a > b);   // Greater than: false
print(a >= b);  // Greater than or equal: false
print(a == b);  // Equal: false
print(a != b);  // Not equal: true
print(x == 3.14); // Float equality: true

// Mathematical functions
print(abs(-5));      // Absolute value of int: 5
print(abs(-2.5));    // Absolute value of float: 2.5
print(sqrt(16));     // Square root: 4.0
print(sqrt(2.25));   // Square root: 1.5

// ===================== VECTORS =====================
print(2000);  // Section marker for vectors

// Vector creation
vector v1;
v1 := [1, 2, 3, 4, 5];
print(v1);

// Vector dimension
int d;
d := dim(v1);
print(d);  // Should be 5

print(v1[0]);  // First element: 1

v1[2] := 10;

// Vector with mixed types
vector v_mixed;

print(v_mixed);

// Vector arithmetic
vector v2;
v2 := [5, 4, 3, 2, 1];
print(v1 + v2);  // [6, 6, 13, 6, 6]
print(v1 - v2);  // [-4, -2, 7, 2, 4]

// Scalar multiplication
print(v1 * 2);   // [2, 4, 20, 8, 10]
print(0.5 * v2); // [2.5, 2.0, 1.5, 1.0, 0.5]

print(v1 dot v2);  // Dot product: 5*1 + 4*2 + 3*10 + 2*4 + 1*5 = 5+8+30+8+5 = 56
print(mag(v1));      // Magnitude: sqrt(1² + 2² + 10² + 4² + 5²) = sqrt(146) ≈ 12.08

// Vector angle
vector unit_x;
unit_x := [1, 0, 0];
vector unit_y;
unit_y := [0, 1, 0];
print(angle(unit_x, unit_y));  // π/2 radians (90 degrees)

vector row;
row := m1[1];

matrix m2;
m2 := [[9, 8, 7], [6, 5, 4], [3, 2, 1]];

ma := [[1, 2], [3, 4]];

mb := [[5, 6], [7, 8]];

// Matrix-vector multiplication
vector v;
v := [1, 2];
vector result;

// Vector-matrix multiplication

// Matrix operations
print(trans(m1));  // Transpose

matrix det_test;
det_test := [[4, 7], [2, 6]];

matrix identity;

matrix verify_inv;

// ===================== ALGORITHM EXAMPLE =====================
print(5000);  // Section marker for algorithm

// Computing eigenvalues of a 2x2 matrix
matrix m;
m := [[3, 2], [1, 4]];

// Calculate trace and determinant
float tr;
float det_val;
tr := trace(m);
det_val := det(m);

// For a 2x2 matrix, eigenvalues = (tr ± sqrt(tr² - 4*det))/2
float discriminant;
discriminant := tr*tr - 4.0*det_val;

float lambda1;
float lambda2;
if (discriminant >= 0.0) {
    lambda1 := (tr + sqrt(discriminant)) / 2.0;
    lambda2 := (tr - sqrt(discriminant)) / 2.0;
    print(lambda1);  // Should be approx. 5
    print(lambda2);  // Should be approx. 2
} else {
    print(0);  // Signal complex eigenvalues
}

// Verify that the product of eigenvalues equals the determinant
float prod_eigenvalues;
prod_eigenvalues := lambda1 * lambda2;
print(prod_eigenvalues);  // Should equal det_val (10)
print(det_val);
