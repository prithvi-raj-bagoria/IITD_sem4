// Advanced operator test cases

// Arithmetic operations with precedence and associativity
int complex_expr1 := 2 + 3 * 4 + 5;
print(complex_expr1);  // Should be 19

int complex_expr2 := (2 + 3) * (4 + 5);
print(complex_expr2);  // Should be 45

int complex_expr3 := 2 ** 3 ** 2;  // Right-associative: 2 ** (3 ** 2) = 2 ** 9 = 512
print(complex_expr3);

float complex_expr4 := 10 / 4;  // Int division coerced to float
print(complex_expr4);  // Should be 2.5

// Unary operations with precedence
int neg_pow := -2 ** 3;  // Should be -(2^3) = -8, not (-2)^3 = -8
print(neg_pow);

// Mixed type arithmetic
int i := 5;
float f := 2.5;
float mixed := i * f;  // 12.5
print(mixed);

// Boolean logic with short-circuit evaluation
bool logic1 := true && false || true;
print(logic1);  // Should be true

bool logic2 := false || true && false;
print(logic2);  // Should be false 

// Complex comparison chains
bool comp := 5 > 3 && 2 < 4 || 7 <= 6;
print(comp);  // Should be true

// Nested expressions
float nested := (3 + 4 * (2 - 1)) / (5 - 2 ** 2);
print(nested);  // (3 + 4) / (5 - 4) = 7
